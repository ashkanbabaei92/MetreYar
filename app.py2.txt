def _pandas_dtype(t: type) -> str:
    if t == str: return "object"
    if t == float: return "float64"
    if t == int: return "int64"
    if t == bool: return "bool"
    return "object"


def make_empty_df(schema: Dict[str, type]) -> pd.DataFrame:
    return pd.DataFrame({col: pd.Series(dtype=_pandas_dtype(t)) for col, t in schema.items()})

    _ALIAS_REVERSE: Dict[str, str] = {}

    @classmethod
    def _build_reverse(cls) -> None:
        if cls._ALIAS_REVERSE: return
        for canonical, aliases in cls.ALIASES.items():
            for alias in aliases:
                cls._ALIAS_REVERSE[cls._normalize_text(alias)] = canonical

    @classmethod
    def fa_to_en_digits(cls, text: str) -> str:
        if not text: return ""
        result = []
        for ch in str(text):
            idx = cls._FA_DIGITS.find(ch)
            if idx >= 0: result.append(cls._EN_DIGITS[idx])
            elif '\u0660' <= ch <= '\u0669': result.append(str(ord(ch) - 0x0660))
            else: result.append(ch)
        return "".join(result)

    @classmethod
    def en_to_fa_digits(cls, text: str) -> str:
        if not text: return ""
        return "".join(cls._FA_DIGITS[int(ch)] if ch.isdigit() else ch for ch in str(text))

    @classmethod
    def _normalize_text(cls, text: str) -> str:
        if not text: return ""
        t = str(text).strip().translate(cls._CHAR_MAP)
        t = cls.fa_to_en_digits(t).lower()
        t = t.replace("\u200c", " ").replace("\u200b", "")
        t = t.replace("\u200d", "").replace("\u200e", "").replace("\u200f", "")
        t = re.sub(r'[\u064B-\u065F\u0670]', '', t)
        t = re.sub(r'[-_/\\.,;:!?(){}[\]"\'«»]+', ' ', t)
        return re.sub(r'\s+', ' ', t).strip()

    @classmethod
    def normalize(cls, text: str) -> str:
        return cls._normalize_text(text)

    @classmethod
    def resolve_alias(cls, text: str) -> Optional[str]:
        cls._build_reverse()
        norm = cls._normalize_text(text)
        if norm in cls._ALIAS_REVERSE: return cls._ALIAS_REVERSE[norm]
        words = norm.split()
        for w in words:
            if w in cls._ALIAS_REVERSE: return cls._ALIAS_REVERSE[w]
        for i in range(len(words) - 1):
            pair = f"{words[i]} {words[i + 1]}"
            if pair in cls._ALIAS_REVERSE: return cls._ALIAS_REVERSE[pair]
        return None

    @classmethod
    def detect_category(cls, text: str) -> Tuple[str, float]:
        norm = cls._normalize_text(text)
        if not norm: return ("unknown", 0.0)

        if any(k in norm for k in ["wall foundation", "wf_", "footing", "پی", "فونداسیون", "شالوده"]):
            return ("foundation", 0.95)

        alias = cls.resolve_alias(norm)
        if alias: return (alias, 0.95)

        cls._build_reverse()
        best_match = None
        best_score = 0.0
        for alias_norm, canonical in cls._ALIAS_REVERSE.items():
            if len(alias_norm) < 2: continue
            if alias_norm in norm:
                score = len(alias_norm) / max(len(norm), 1)
                if score > best_score:
                    best_score = score
                    best_match = canonical
        if best_match and best_score > 0.3:
            return (best_match, min(best_score + 0.3, 0.9))
        return ("unknown", 0.0)

    @classmethod
    def detect_structural_group(cls, text: str) -> str:
        cat, _ = cls.detect_category(text)
        group_map = {
            "foundation": "پی", "column": "ستون", "beam": "تیر",
            "slab": "سقف", "block_joist": "سقف تیرچه", "eps_block": "سقف تیرچه",
            "tie_beam": "کلاف", "stair": "پله", "wall": "دیوار",
        }
        return group_map.get(cat, "سایر")

    @classmethod
    def safe_float(cls, value: Any) -> float:
        if value is None: return 0.0
        if isinstance(value, (int, float)): return float(value)
        t = cls.fa_to_en_digits(str(value).strip())
        t = t.replace(',', '.').replace('٫', '.')
        t = re.sub(r'[^\d.\-+eE]', '', t)
        if not t or t in ('.', '-', '+'): return 0.0
        try: return float(t)
        except ValueError: return 0.0

    @classmethod
    def parse_number_with_unit(cls, text: str) -> Tuple[Optional[float], Optional[str]]:
        if not text: return (None, None)
        t = cls.fa_to_en_digits(str(text).strip())
        m = re.match(r'^([+-]?\d+(?:[.,]\d+)?)\s*([a-zA-Z³²%°]+.*)?$', t)
        if m:
            try: val = float(m.group(1).replace(',', '.'))
            except ValueError: return (None, None)
            unit = m.group(2).strip() if m.group(2) else None
            return (val, unit)
        try: return (float(t.replace(',', '.')), None)
        except ValueError: return (None, None)

# Revit Element Detector (فوق‌حرفه‌ای)
# ──────────────────────────────────────────────
class RevitElementDetector:
    """Detects Revit element type from Mark/Name based on Iranian structural naming conventions.
    
    Structural Naming Conventions:
    - BP  = Beam of Foundation Pile (تیر شمعی/آرماتور پی)
    - LG  = Longitudinal Girder (تیر طولی)
    - TR  = Transverse Rebar (میلگرد عرضی)
    - CT  = Cross Tie / Coupling Tie (کلاف)
    - STR = Stair (پله)
    - JST = Joist Stirrup (خاموت تیرچه)
    - JT  = Joist Tie (آرماتور اتصال تیرچه)
    - TM  = Temperature Mesh (میلگرد حرارتی/توری)
    - C   = Column (ستون) - e.g., C1, C2
    - L   = Longitudinal (طولی)
    - T   = Transverse (عرضی) - e.g., T(A-B)
    - ST  = Stirrup (خاموت) - e.g., BP4-ST1-ST
    - B   = Bottom (پایین)
    - Top / T = Top (بالا)
    - R   = Reinforcement / Additional (تقویتی)
    - ADD = Additional (تقویتی)
    - Main = Main bar (اصلی)
    - Starter = Anchor bar (انتظار)
    - F = Foundation/Field
    """

    # Prefix → (structural_group, sub_group)
    PREFIX_MAP: Dict[str, Tuple[str, str]] = {
        "BP": ("پی", "آرماتور پی"),
        "LG": ("تیر", "آرماتور طولی تیر"),
        "TR": ("تیر", "آرماتور عرضی تیر"),
        "CT": ("کلاف", "کلاف"),
        "STR": ("پله", "آرماتور پله"),
        "JST": ("سقف تیرچه", "خاموت تیرچه"),
        "JT": ("سقف تیرچه", "آرماتور اتصال تیرچه"),
        "TM": ("سقف", "میلگرد حرارتی"),
        "C1": ("ستون", "آرماتور ستون"),
        "C2": ("ستون", "آرماتور ستون"),
        "C3": ("ستون", "آرماتور ستون"),
        "C4": ("ستون", "آرماتور ستون"),
        "C5": ("ستون", "آرماتور ستون"),
    }

    # Wall Foundation Marks (L for longitudinal, T for transverse)
    FOUNDATION_MARK_PATTERNS = [
        r'^L\(\s*\d+\s*[-–]\s*\d+\s*\)',   # L(1-2), L(2-3)
        r'^T\(\s*[A-Z]+\s*[-–]\s*[A-Z]+\s*\)',  # T(A-B), T(B-C)
    ]

    @classmethod
    def detect_element(cls, mark: str, family: str = "", table_title: str = "") -> Dict:
        """Detect element structure from Mark/Name.
        Returns dict with: structural_group, sub_group, element_type, element_position
        """
        result = {
            "structural_group": "سایر",
            "sub_group": "نامشخص",
            "element_type": "unknown",
            "element_position": "",
        }
        
        if not mark and not family:
            return result
        
        combined = f"{mark} {family}".strip()
        combined_lower = combined.lower()
        table_lower = table_title.lower() if table_title else ""

        # ==========================================
        # 1. FOUNDATION (Wall Foundation / L(...)/T(...))
        # ==========================================
        for pattern in cls.FOUNDATION_MARK_PATTERNS:
            if re.search(pattern, mark, re.IGNORECASE):
                result["structural_group"] = "پی"
                result["sub_group"] = "پی نواری"
                result["element_type"] = "wall_foundation"
                if mark.upper().startswith("L"):
                    result["element_position"] = "طولی"
                elif mark.upper().startswith("T"):
                    result["element_position"] = "عرضی"
                return result

        if "wall foundation" in combined_lower or "wf_" in combined_lower:
            result["structural_group"] = "پی"
            result["sub_group"] = "پی نواری"
            result["element_type"] = "wall_foundation"
            return result

        if "foundation slab" in combined_lower or "100mm foundation" in combined_lower or "lc-01" in combined_lower:
            result["structural_group"] = "پی"
            result["sub_group"] = "بتن مگر"
            result["element_type"] = "lean_concrete"
            return result

        if "footing" in combined_lower:
            result["structural_group"] = "پی"
            result["sub_group"] = "پی منفرد"
            result["element_type"] = "footing"
            return result

        # ==========================================
        # 2. Extract Prefix (e.g., "BP4-ST1-ADD-B1" -> "BP")
        # ==========================================
        prefix_match = re.match(r'^([A-Z]{1,4})(\d*)', mark.upper().strip())
        prefix = prefix_match.group(1) if prefix_match else ""
        
        # Check for exact prefix match
        for key, (grp, sub) in cls.PREFIX_MAP.items():
            if mark.upper().startswith(key):
                result["structural_group"] = grp
                result["sub_group"] = sub
                result["element_type"] = key.lower()
                break

        # ==========================================
        # 3. Detect Position (Top / Bottom / Longitudinal / Stirrup)
        # ==========================================
        mark_upper = mark.upper()
        
        # Stirrup detection (خاموت)
        if "-ST" in mark_upper or "STIRRUP" in combined_lower or "خاموت" in combined:
            if result["structural_group"] == "سایر":
                if "C1-" in mark_upper or "C2-" in mark_upper:
                    result["structural_group"] = "ستون"
            result["sub_group"] = "خاموت " + result["structural_group"] if result["structural_group"] != "سایر" else "خاموت"
            result["element_position"] = "خاموت"
            return result

        # ADD = Additional / Reinforcement (تقویتی)
        if "-ADD-" in mark_upper or "ADD " in mark_upper:
            if "B" in mark_upper.split("-")[-1]:
                result["element_position"] = "تقویتی پایین"
            elif "T" in mark_upper.split("-")[-1]:
                result["element_position"] = "تقویتی بالا"
            else:
                result["element_position"] = "تقویتی"
            result["sub_group"] = "آرماتور تقویتی " + result["structural_group"]
            return result

        # R1, R2, R3... = Reinforcement bars (تقویتی)
        r_match = re.search(r'-R(\d+)-([BT])', mark_upper)
        if r_match:
            r_num = r_match.group(1)
            pos = r_match.group(2)
            result["element_position"] = f"تقویتی {'پایین' if pos == 'B' else 'بالا'} R{r_num}"
            result["sub_group"] = f"آرماتور تقویتی {result['structural_group']}"
            return result

        # Main-B / Main-T
        if "-MAIN-B" in mark_upper:
            result["element_position"] = "اصلی پایین"
            result["sub_group"] = "آرماتور اصلی " + result["structural_group"]
        elif "-MAIN-T" in mark_upper:
            result["element_position"] = "اصلی بالا"
            result["sub_group"] = "آرماتور اصلی " + result["structural_group"]
        elif "-STARTER" in mark_upper or "STARTER-" in mark_upper:
            result["element_position"] = "انتظار"
            result["sub_group"] = "آرماتور انتظار " + result["structural_group"]
        elif "-TR-" in mark_upper:
            if mark_upper.endswith("-B"):
                result["element_position"] = "عرضی پایین"
            elif mark_upper.endswith("-T"):
                result["element_position"] = "عرضی بالا"
            else:
                result["element_position"] = "عرضی"
            result["sub_group"] = "آرماتور عرضی " + result["structural_group"]
        elif mark_upper.endswith("-B") or mark_upper.endswith("_B"):
            result["element_position"] = "پایین"
            if "آرماتور" not in result["sub_group"]:
                result["sub_group"] = "آرماتور اصلی " + result["structural_group"]
        elif mark_upper.endswith("-T") or mark_upper.endswith("_T"):
            result["element_position"] = "بالا"
            if "آرماتور" not in result["sub_group"]:
                result["sub_group"] = "آرماتور اصلی " + result["structural_group"]
        elif mark_upper.endswith("-L") or "-L-" in mark_upper:
            result["element_position"] = "طولی"
            if "آرماتور" not in result["sub_group"]:
                result["sub_group"] = "آرماتور طولی " + result["structural_group"]

        # ==========================================
        # 4. Table title override (if no prefix matched)
        # ==========================================
        if result["structural_group"] == "سایر" and table_title:
            if "پی" in table_title or "foundation" in table_lower:
                result["structural_group"] = "پی"
                result["sub_group"] = "پی"
            elif "ستون" in table_title or "column" in table_lower:
                result["structural_group"] = "ستون"
                result["sub_group"] = "ستون"
            elif "تیر" in table_title or "beam" in table_lower:
                result["structural_group"] = "تیر"
                result["sub_group"] = "تیر"
            elif "پله" in table_title or "stair" in table_lower:
                result["structural_group"] = "پله"
                result["sub_group"] = "پله"
            elif "سقف" in table_title or "slab" in table_lower or "joist" in table_lower:
                result["structural_group"] = "سقف"
                result["sub_group"] = "سقف"

        # ==========================================
        # 5. Family-based detection (Structural Columns / Framing)
        # ==========================================
        if "structural columns" in combined_lower or "concrete-rectangular-column" in combined_lower or "concrete-square-column" in combined_lower:
            result["structural_group"] = "ستون"
            result["sub_group"] = "بتن ستون"
            result["element_type"] = "column"
        elif "structural framing" in combined_lower or "concrete-rectangular beam" in combined_lower:
            result["structural_group"] = "تیر"
            result["sub_group"] = "بتن تیر"
            result["element_type"] = "beam"
        elif "cast-in-place stair" in combined_lower or "monolithic stair" in combined_lower or "monolithic run" in combined_lower:
            result["structural_group"] = "پله"
            result["sub_group"] = "بتن پله"
            result["element_type"] = "stair"
        elif "floor" in combined_lower and "slab" in combined_lower:
            if "topping" in combined_lower:
                result["structural_group"] = "سقف"
                result["sub_group"] = "بتن روی سقف"
            else:
                result["structural_group"] = "سقف"
                result["sub_group"] = "بتن دال"
        elif "تیرچه و یونولیت" in combined or "joist" in combined_lower:
            result["structural_group"] = "سقف تیرچه"
            if "یونولیت" in combined or "eps" in combined_lower:
                result["sub_group"] = "بلوک یونولیت"
                result["element_type"] = "eps_block"
            elif "پاشنه" in combined:
                result["sub_group"] = "بتن پاشنه تیرچه"
                result["element_type"] = "joist_concrete"
            else:
                result["sub_group"] = "تیرچه"
                result["element_type"] = "joist"

        return result

    @classmethod
    def extract_diameter(cls, text: str) -> int:
        """Extract rebar diameter from text (Φ18, Φ10, 18mm, A-I-8mm)."""
        if not text: return 0
        # Try φ or Φ format
        m = re.search(r'[φΦ]\s*(\d{1,2})', text)
        if m:
            try: return int(m.group(1))
            except: pass
        # Try "18 mm" format
        m = re.search(r'(\d{1,2})\s*mm', text, re.IGNORECASE)
        if m:
            try: return int(m.group(1))
            except: pass
        # Try "Rebar - Φ8" format
        m = re.search(r'rebar\s*-\s*[φΦ]?(\d{1,2})', text, re.IGNORECASE)
        if m:
            try: return int(m.group(1))
            except: pass
        # Try "A-I-8mm" format (A1 - 8mm)
        m = re.search(r'a[\-\s]*i[\-\s]*(\d{1,2})\s*mm', text, re.IGNORECASE)
        if m:
            try: return int(m.group(1))
            except: pass
        return 0

    @classmethod
    def is_a1_rebar(cls, text: str) -> bool:
        """Check if this is A1 (plain/round) rebar (میلگرد ساده)."""
        if not text: return False
        t = text.lower()
        return ("a-i-" in t or "a1" in t or "fy2400" in t or 
                "s240" in t or "plain" in t or "ساده" in text)

    @classmethod
    def is_ajdar_rebar(cls, text: str) -> bool:
        """Check if this is deformed rebar (میلگرد آجدار)."""
        if not text: return False
        t = text.lower()
        return ("a3" in t or "fy4000" in t or "s400" in t or 
                "deformed" in t or "آجدار" in text or 
                bool(re.search(r'[φΦ]\d', text)))

# Fast Fuzzy Matcher
# ──────────────────────────────────────────────
class FuzzyMatcher:
    @staticmethod
    def best_match(query: str, candidates: List[str], threshold: float = 0.35) -> Optional[Tuple[str, float]]:
        if not query or not candidates: return None
        query_words = set(query.split())
        if not query_words: return None

        best_cand = None
        best_score = 0.0
        for cand in candidates:
            cand_words = set(cand.split())
            common = query_words.intersection(cand_words)
            if not common: continue
            overlap = len(common) / max(len(query_words), len(cand_words))
            seq = difflib.SequenceMatcher(None, query, cand).quick_ratio()
            final = max(overlap, seq)
            if final > best_score:
                best_score = final
                best_cand = cand
        if best_cand and best_score >= threshold:
            return (best_cand, best_score)
        return None

    @staticmethod
    def find_all_containing(query: str, candidates: List[Dict], keyword_field: str = "norm_desc",
                            keywords: List[str] = None, exclude_keywords: List[str] = None) -> List[Dict]:
        """Find all candidates that contain ALL required keywords AND NONE of excluded."""
        results = []
        keywords = keywords or []
        exclude_keywords = exclude_keywords or []
        for c in candidates:
            desc = c.get(keyword_field, "")
            if not all(k in desc for k in keywords): continue
            if any(k in desc for k in exclude_keywords): continue
            results.append(c)
        return results

class DatabaseManager:
    """مدیر پایگاه داده SQLite برای ذخیره‌سازی دائمی پروژه‌های دفتر فنی.
    
    ویژگی‌های حرفه‌ای:
    ─────────────────────────────────────────
    • پروژه‌ها با متادیتای کامل (کارفرما، پیمانکار، مبلغ قرارداد)
    • فهرست بها با پشتیبانی از ردیف‌های ستاره‌دار (آنالیز بها)
    • جداول Revit با ذخیره‌سازی JSON کامل
    • قوانین Mapping با اولویت‌بندی
    • تنظیمات پرت به تفکیک دسته
    • ضرایب پیمان (بالاسری، منطقه‌ای، طبقات، ارتفاع)
    • آنالیز بها برای ردیف‌های ستاره‌دار
    • ذخیره تاریخچه صورت وضعیت‌ها (مقایسه پیشرفت)
    • شاخص‌های تعدیل بانک مرکزی (نشریه ۱۰۰)
    • بنچمارک پروژه‌های مشابه
    • لاگ عملیات (audit trail)
    """

    def __init__(self, db_path: str = DB_NAME):
        self.db_path = db_path
        self.conn: Optional[sqlite3.Connection] = None
        self._init_db()

    def _init_db(self) -> None:
        """راه‌اندازی اتصال با تنظیمات بهینه SQLite."""
        self.conn = sqlite3.connect(
            self.db_path,
            check_same_thread=False,
            detect_types=sqlite3.PARSE_DECLTYPES
        )
        # Performance & safety pragmas
        self.conn.execute("PRAGMA journal_mode=WAL")
        self.conn.execute("PRAGMA foreign_keys=ON")
        self.conn.execute("PRAGMA synchronous=NORMAL")
        self.conn.execute("PRAGMA cache_size=-64000")  # 64MB cache
        self.conn.execute("PRAGMA temp_store=MEMORY")
        self.conn.row_factory = sqlite3.Row  # Named column access
        self._create_tables()
        self._migrate_if_needed()

    def _migrate_if_needed(self):
        """
        بررسی و ثبت نسخه Schema دیتابیس.

        _create_tables فقط جداول را ایجاد می‌کند.
        تعیین نسخه Schema در این تابع انجام می‌شود.
        """

        current_version = None

        try:
            row = self.conn.execute(
                """
                SELECT value
                FROM system_meta
                WHERE key = 'schema_version'
                LIMIT 1
                """
            ).fetchone()

            if row:
                current_version = row[0]

        except Exception as exc:
            logger.warning(
                "Could not read database schema version: %s",
                exc
            )

        # ---------------------------------------------------------
        # اگر دیتابیس از قبل همین نسخه را دارد،
        # هیچ کاری لازم نیست.
        # ---------------------------------------------------------
        if current_version == self.SCHEMA_VERSION:
            return

        try:
            # -----------------------------------------------------
            # Migration فعلی additive است.
            #
            # داده‌های Semantic داخل raw_data نگهداری می‌شوند،
            # بنابراین داده‌های قبلی حذف یا بازسازی نمی‌شوند.
            # -----------------------------------------------------
            self.conn.execute(
                """
                INSERT INTO system_meta (
                    key,
                    value
                )
                VALUES (?, ?)
                ON CONFLICT(key)
                DO UPDATE SET
                    value = excluded.value,
                    updated_at = datetime('now','localtime')
                """,
                (
                    "schema_version",
                    self.SCHEMA_VERSION,
                ),
            )

            self.conn.commit()

            logger.info(
                "Database schema migrated: %s -> %s",
                current_version,
                self.SCHEMA_VERSION,
            )

        except Exception as exc:
            self.conn.rollback()

            logger.exception(
                "Database migration failed: %s",
                exc,
            )

            raise

def _migrate_if_needed(self):

    current_version = None

    try:
        row = self.conn.execute(
            """
            SELECT value
            FROM system_meta
            WHERE key = 'schema_version'
            LIMIT 1
            """
        ).fetchone()

        if row:
            current_version = row[0]

    except Exception:
        current_version = None

    if current_version == self.SCHEMA_VERSION:
        return

    try:

        # ----------------------------------------------------
        # Migration های additive
        # ----------------------------------------------------
        #
        # چون semantic fields داخل raw_data نیز نگهداری می‌شوند،
        # نیازی به حذف یا بازسازی داده‌های قبلی نیست.
        #

        self.conn.execute(
            """
            INSERT INTO system_meta(key, value)
            VALUES('schema_version', ?)
            ON CONFLICT(key)
            DO UPDATE SET value=excluded.value
            """,
            (self.SCHEMA_VERSION,),
        )

        self.conn.commit()

        logger.info(
            "Database schema migrated: %s -> %s",
            current_version,
            self.SCHEMA_VERSION,
        )

    except Exception as exc:

        self.conn.rollback()

        logger.exception(
            "Database migration failed: %s",
            exc
        )

        raise

    def update_project(self, project_id: int, **kwargs) -> None:
        """به‌روزرسانی متادیتای پروژه."""
        if not kwargs:
            return
        fields = []
        values = []
        allowed = {"name", "description", "price_list_year", "client_name",
                   "contractor_name", "consultant_name", "project_location",
                   "project_area_m2", "floors_count", "contract_amount",
                   "contract_date", "contract_number", "start_date", "end_date",
                   "project_type", "structure_type", "status"}
        for k, v in kwargs.items():
            if k in allowed:
                fields.append(f"{k}=?")
                values.append(v)
        if not fields:
            return
        fields.append("updated_at=datetime('now','localtime')")
        values.append(project_id)
        query = f"UPDATE projects SET {', '.join(fields)} WHERE id=?"
        self.conn.execute(query, values)
        self.conn.commit()
        self.log_action(project_id, "UPDATE_PROJECT", ", ".join(kwargs.keys()))

def save_revit_data(
    self,
    project_id: int,
    file_id: int,
    records: List[Dict]
) -> int:
    """
    ذخیره ردیف‌های Revit در دیتابیس.

    نکات مهم:
    - raw_data تمام داده‌های خام و Semantic را حفظ می‌کند.
    - اگر raw_data قبلاً dict باشد، فقط یک بار JSON می‌شود.
    - اگر raw_data وجود نداشته باشد، کل rec به عنوان raw_data ذخیره می‌شود.
    - Quantity خالی یا نامشخص هرگز به 1 تبدیل نمی‌شود.
    - source_table و سایر اطلاعات traceability حفظ می‌شوند.
    """

    c = self.conn.cursor()
    count = 0

    for rec in records:

        # ---------------------------------------------------------
        # 1) حفظ کامل داده خام Revit / Dynamo / Semantic
        # ---------------------------------------------------------
        raw_data_value = rec.get("raw_data", "")

        if isinstance(raw_data_value, dict):
            raw_data_value = json.dumps(
                raw_data_value,
                ensure_ascii=False,
                default=str,
            )

        elif isinstance(raw_data_value, list):
            raw_data_value = json.dumps(
                raw_data_value,
                ensure_ascii=False,
                default=str,
            )

        elif raw_data_value is None or raw_data_value == "":
            raw_data_value = json.dumps(
                rec,
                ensure_ascii=False,
                default=str,
            )

        else:
            raw_data_value = str(raw_data_value)

        # ---------------------------------------------------------
        # 2) ذخیره رکورد
        # ---------------------------------------------------------
        c.execute(
            """
            INSERT INTO revit_data
            (
                file_id,
                project_id,
                row_index,
                raw_data,
                raw_name,
                normalized_name,
                source_table,
                detected_category,
                detected_group,
                sub_group,
                element_type,
                element_position,
                floor_level,
                volume,
                area,
                length,
                width,
                height,
                depth,
                count,
                weight,
                diameter,
                bar_length,
                number_of_bars,
                thickness,
                rebar_total_weight
            )
            VALUES (
                ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?,
                ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?
            )
            """,
            (
                file_id,
                project_id,

                rec.get("row_index", 0),

                # مهم:
                # اینجا دیگر json.dumps مستقیم روی rec.get()
                # انجام نمی‌شود.
                raw_data_value,

                rec.get("raw_name", ""),
                rec.get("normalized_name", ""),

                # تفکیک جدول/فایل Revit
                rec.get("source_table", ""),

                rec.get("detected_category", ""),
                rec.get("detected_group", ""),
                rec.get("sub_group", ""),
                rec.get("element_type", ""),
                rec.get("element_position", ""),
                rec.get("floor_level", ""),

                rec.get("volume", 0),
                rec.get("area", 0),
                rec.get("length", 0),
                rec.get("width", 0),
                rec.get("height", 0),
                rec.get("depth", 0),

                # مهم:
                # این مقدار مستقیماً از importer می‌آید.
                # اینجا هیچ fallback به 1 نداریم.
                rec.get("count", 0),

                rec.get("weight", 0),

                rec.get("diameter", 0),
                rec.get("bar_length", 0),
                rec.get("number_of_bars", 0),
                rec.get("thickness", 0),
                rec.get("rebar_total_weight", 0),
            ),
        )

        count += 1

    self.conn.commit()

    return count

            return {"error": "یکی از نتایج یافت نشد"}

        s1 = r1.get("summary", {})
        s2 = r2.get("summary", {})
        return {
            "total_price_1": s1.get("total_price", 0),
            "total_price_2": s2.get("total_price", 0),
            "diff_amount": s2.get("total_price", 0) - s1.get("total_price", 0),
            "diff_percent": (
                (s2.get("total_price", 0) - s1.get("total_price", 0))
                / max(s1.get("total_price", 1), 1) * 100
            ),
            "items_1": s1.get("total_items", 0),
            "items_2": s2.get("total_items", 0),

class PriceListImporter:
    """Smart importer for Iranian construction price lists (فهرست بها).
    Handles zero-padded codes (070101 vs 70101), merged cells, and multi-sheet files.
    """

    CODE_PATTERNS = [
        r"کد\s*ردیف", r"شماره\s*ردیف", r"شماره", r"کد", r"ردیف",
        r"item[\s_-]*code", r"code", r"row[\s_-]*number",
    ]
    DESC_PATTERNS = [
        r"شرح", r"شرح\s*عملیات", r"شرح\s*ردیف", r"توضیح",
        r"description", r"desc",
    ]
    UNIT_PATTERNS = [r"واحد", r"unit"]
    PRICE_PATTERNS = [
        r"بهای\s*واحد", r"بهاي\s*واحد", r"قیمت\s*واحد", r"قیمت", r"فی",
        r"بها", r"unit[\s_-]*price", r"price", r"amount",
    ]
    CHAPTER_PATTERNS = [r"فصل", r"chapter", r"chap"]

    @classmethod
    def detect_year(cls, filename: str, df: Optional[pd.DataFrame] = None) -> str:
        text = TextNormalizer.fa_to_en_digits(filename)
        m = re.search(r'(1[34]\d{2})', text)
        if m: return m.group(1)
        if df is not None:
            for col in df.columns[:5]:
                for val in df[col].head(10).astype(str):
                    val = TextNormalizer.fa_to_en_digits(val)
                    m = re.search(r'(1[34]\d{2})', val)
                    if m: return m.group(1)
        return ""

    @classmethod
    def _match_column(cls, col_name: str, patterns: List[str]) -> bool:
        norm = TextNormalizer.normalize(col_name)
        for pat in patterns:
            if re.search(pat, norm, re.IGNORECASE): return True
        return False

    @classmethod
    def _find_header_row(cls, df_raw: pd.DataFrame, max_rows: int = 20) -> int:
        for i in range(min(max_rows, len(df_raw))):
            row_values = [TextNormalizer.normalize(str(v)) for v in df_raw.iloc[i]]
            score = 0
            for val in row_values:
                if not val or val == 'nan': continue
                for pats in [cls.CODE_PATTERNS, cls.DESC_PATTERNS,
                             cls.UNIT_PATTERNS, cls.PRICE_PATTERNS]:
                    if any(re.search(p, val, re.IGNORECASE) for p in pats):
                        score += 1
                        break
            if score >= 2: return i
        return 0

    @classmethod
    def _identify_columns(cls, columns: List[str]) -> Dict[str, Optional[int]]:
        result = {"code": None, "desc": None, "unit": None, "price": None, "chapter": None}
        for i, col in enumerate(columns):
            if result["code"] is None and cls._match_column(col, cls.CODE_PATTERNS): result["code"] = i
            elif result["desc"] is None and cls._match_column(col, cls.DESC_PATTERNS): result["desc"] = i
            elif result["unit"] is None and cls._match_column(col, cls.UNIT_PATTERNS): result["unit"] = i
            elif result["price"] is None and cls._match_column(col, cls.PRICE_PATTERNS): result["price"] = i
            elif result["chapter"] is None and cls._match_column(col, cls.CHAPTER_PATTERNS): result["chapter"] = i
        return result

    @classmethod
    def import_excel(cls, file_data: io.BytesIO, filename: str = "") -> Tuple[pd.DataFrame, Dict]:
        info = {
            "filename": filename, "year": "", "sheets_processed": [],
            "total_rows": 0, "valid_rows": 0, "starred_rows": 0,
            "errors": [], "warnings": [],
        }
        try: xls = pd.ExcelFile(file_data, engine="openpyxl")
        except Exception as e:
            info["errors"].append(f"خطا در باز کردن فایل: {e}")
            return make_empty_df(PRICE_LIST_SCHEMA), info

        all_items = []
        year_detected = ""

        for sheet_name in xls.sheet_names:
            try: df_raw = pd.read_excel(xls, sheet_name=sheet_name, header=None, dtype=str)
            except Exception as e:
                info["warnings"].append(f"شیت '{sheet_name}' قابل خواندن نیست: {e}")
                continue
            if df_raw.empty or len(df_raw) < 2: continue
            try: df_raw = df_raw.ffill(axis=0)
            except Exception: pass

            header_row = cls._find_header_row(df_raw)
            headers = [str(v) for v in df_raw.iloc[header_row]]
            col_map = cls._identify_columns(headers)

            if col_map["code"] is None and col_map["desc"] is None:
                info["warnings"].append(f"شیت '{sheet_name}': ستون‌های شناخته‌شده پیدا نشد")
                continue

            info["sheets_processed"].append(sheet_name)
            df_data = df_raw.iloc[header_row + 1:].reset_index(drop=True)

            if not year_detected:
                year_detected = cls.detect_year(filename, df_raw.iloc[:header_row + 1])

            current_chapter = ""
            current_chapter_title = ""

            for idx, row in df_data.iterrows():
                info["total_rows"] += 1
                code_val = str(row.iloc[col_map["code"]]).strip() if col_map["code"] is not None else ""
                desc_val = str(row.iloc[col_map["desc"]]).strip() if col_map["desc"] is not None else ""
                unit_val = str(row.iloc[col_map["unit"]]).strip() if col_map["unit"] is not None else ""
                price_val = str(row.iloc[col_map["price"]]).strip() if col_map["price"] is not None else "0"
                chapter_val = str(row.iloc[col_map["chapter"]]).strip() if col_map["chapter"] is not None else ""

                code_val = TextNormalizer.fa_to_en_digits(code_val)
                price_val = TextNormalizer.fa_to_en_digits(price_val)

                if code_val in ("", "nan", "None") and desc_val in ("", "nan", "None"): continue
                if chapter_val and chapter_val not in ("", "nan", "None"):
                    current_chapter = TextNormalizer.fa_to_en_digits(chapter_val)
                    current_chapter_title = desc_val if desc_val not in ("", "nan", "None") else ""

                if (unit_val in ("", "nan", "None") and price_val in ("", "nan", "None", "0") and desc_val not in ("", "nan", "None")):
                    ch_match = re.match(r'^(\d{1,2})', code_val)
                    if ch_match and len(code_val) <= 3:
                        current_chapter = code_val
                        current_chapter_title = desc_val
                        continue

                if not re.search(r'\d', code_val): continue
                is_starred = '*' in code_val or '٭' in code_val or '*' in desc_val
                code_val = code_val.replace('*', '').replace('٭', '').strip()
                
                # Standardize 6-digit code with leading zero if needed (e.g. 70101 -> 070101)
                code_clean = re.sub(r'^\d{1,2}\s*-\s*', '', code_val).replace('.', '').strip()
                if len(code_clean) == 5 and code_clean.isdigit():
                    code_clean = "0" + code_clean

                price_float = TextNormalizer.safe_float(price_val)

                item = {
                    "item_code": code_clean,
                    "description": desc_val if desc_val not in ("nan", "None") else "",
                    "unit": unit_val if unit_val not in ("nan", "None") else "",
                    "unit_price": price_float,
                    "chapter": current_chapter or (code_clean[:2] if len(code_clean) >= 2 else ""),
                    "chapter_title": current_chapter_title,
                    "is_starred": is_starred,
                }
                all_items.append(item)
                info["valid_rows"] += 1
                if is_starred: info["starred_rows"] += 1

        info["year"] = year_detected
        if not all_items:
            info["errors"].append("هیچ ردیف معتبری در فایل یافت نشد")
            return make_empty_df(PRICE_LIST_SCHEMA), info

        df = pd.DataFrame(all_items)
        for col, dtype in PRICE_LIST_SCHEMA.items():
            if col not in df.columns: df[col] = None
            try: df[col] = df[col].astype(_pandas_dtype(dtype))
            except Exception: pass
        return df, info

class UnitEngine:
    """
    نرمال‌سازی و کنترل واحدها.
    نکته مهم:
    - هیچ مقداری را خودسرانه تبدیل نمی‌کند مگر واحد مبدأ و مقصد مشخص باشند.
    """

    UNIT_ALIASES = {
        "مترمکعب": "m3",
        "متر مکعب": "m3",
        "m3": "m3",
        "m³": "m3",
        "مترمربع": "m2",
        "متر مربع": "m2",
        "m2": "m2",
        "m²": "m2",
        "متر": "m",
        "m": "m",
        "میلیمتر": "mm",
        "میلی‌متر": "mm",
        "mm": "mm",
        "سانتی‌متر": "cm",
        "cm": "cm",
        "کیلوگرم": "kg",
        "کیلو گرم": "kg",
        "kg": "kg",
        "تن": "ton",
        "ton": "ton",
        "عدد": "ea",
        "دستگاه": "ea",
        "مورد": "ea",
        "ea": "ea",
        "pcs": "ea",
    }

    @classmethod
    def normalize(cls, unit: Any) -> str:
        if unit is None:
            return ""

        text = str(unit).strip().lower()
        text = text.replace(" ", "")
        text = text.replace("‌", "")

        if text in cls.UNIT_ALIASES:
            return cls.UNIT_ALIASES[text]

        return text

    @classmethod
    def compatible(cls, source_unit: Any, target_unit: Any) -> bool:
        s = cls.normalize(source_unit)
        t = cls.normalize(target_unit)

        if not s or not t:
            return True

        if s == t:
            return True

        compatible_groups = [
            {"kg", "ton"},
            {"m", "mm", "cm"},
        ]

        for group in compatible_groups:
            if s in group and t in group:
                return True

        return False

    @classmethod
    def convert(
        cls,
        value: float,
        source_unit: Any,
        target_unit: Any,
    ) -> Optional[float]:

        s = cls.normalize(source_unit)
        t = cls.normalize(target_unit)

        if not s or not t or s == t:
            return float(value)

        # طول
        if s == "mm" and t == "m":
            return float(value) / 1000.0

        if s == "cm" and t == "m":
            return float(value) / 100.0

        if s == "m" and t == "mm":
            return float(value) * 1000.0

        if s == "m" and t == "cm":
            return float(value) * 100.0

        # جرم
        if s == "kg" and t == "ton":
            return float(value) / 1000.0

        if s == "ton" and t == "kg":
            return float(value) * 1000.0

        return None


class RebarQuantityEngine:
    """
    محاسبه مقدار واقعی آرماتور.

    اولویت:
    1) وزن صریح Revit
    2) وزن موجود در schedule
    3) طول کل میلگرد
    4) طول هر میلگرد × تعداد
    5) در صورت نبود Quantity، هرگز مقدار 1 فرض نمی‌شود.
    """

    @staticmethod
    def _num(value: Any) -> Optional[float]:
        try:
            if value is None:
                return None

            text = str(value).strip()

            if not text:
                return None

            text = text.replace(",", "")
            text = text.replace("٬", "")
            text = text.replace("،", ".")

            result = float(text)

            if math.isnan(result) or math.isinf(result):
                return None

            return result

        except Exception:
            return None

    @classmethod
    def calculate(
        cls,
        row: Dict[str, Any],
        diameter: Optional[float] = None,
    ) -> Dict[str, Any]:

        issues: List[str] = []

        explicit_weight = None

        for key in (
            "rebar_total_weight",
            "TotalWeight",
            "Weight",
            "weight",
            "total_weight",
            "وزن کل",
        ):
            if key in row:
                explicit_weight = cls._num(row.get(key))
                if explicit_weight is not None and explicit_weight > 0:
                    return {
                        "weight_kg": explicit_weight,
                        "method": "explicit_weight",
                        "needs_review": False,
                        "issue": "",
                    }

        diameter_value = diameter

        if diameter_value is None:
            for key in (
                "Diameter_mm",
                "Diameter",
                "diameter",
                "BarDiameter",
                "قطر",
            ):
                d = cls._num(row.get(key))
                if d is not None and d > 0:
                    diameter_value = d
                    break

        if not diameter_value:
            return {
                "weight_kg": 0.0,
                "method": "no_diameter",
                "needs_review": True,
                "issue": "قطر آرماتور مشخص نیست",
            }

        unit_weight = REBAR_UNIT_WEIGHT.get(int(round(diameter_value)))

        if not unit_weight:
            unit_weight = (diameter_value ** 2) / 162.0

        # طول کل
        total_length_mm = None

        for key in (
            "TotalLength_mm",
            "Total_Length_mm",
            "TotalBarLength_mm",
            "LengthTotal_mm",
        ):
            val = cls._num(row.get(key))
            if val is not None and val > 0:
                total_length_mm = val
                break

        if total_length_mm is not None:
            total_length_m = total_length_mm / 1000.0
            weight = total_length_m * unit_weight

            return {
                "weight_kg": weight,
                "method": "total_length",
                "needs_review": False,
                "issue": "",
            }

        # طول هر میلگرد
        length_mm = None

        for key in (
            "Length_mm",
            "length_mm",
            "BarLength_mm",
            "bar_length",
            "CutLength_mm",
            "cut_length",
        ):
            val = cls._num(row.get(key))
            if val is not None and val > 0:
                length_mm = val
                break

        if length_mm is None:
            return {
                "weight_kg": 0.0,
                "method": "no_length",
                "needs_review": True,
                "issue": "طول آرماتور مشخص نیست",
            }

        # تعداد
        quantity = None

        for key in (
            "Quantity",
            "quantity",
            "Count",
            "count",
            "Number",
            "number_of_bars",
            "NumberOfBars",
            "Qty",
        ):
            if key in row:
                q = cls._num(row.get(key))

                # نکته بسیار مهم:
                # نبود Quantity یعنی نامشخص، نه یک.
                if q is not None and q > 0:
                    quantity = q
                    break

        if quantity is None:
            issues.append(
                "تعداد آرماتور در داده Revit مشخص نشده است؛ "
                "مقدار 1 فرض نشده است."
            )

            return {
                "weight_kg": 0.0,
                "method": "missing_quantity",
                "needs_review": True,
                "issue": " | ".join(issues),
            }

        total_length_m = (length_mm / 1000.0) * quantity
        weight = total_length_m * unit_weight

        return {
            "weight_kg": weight,
            "method": "length_x_quantity",
            "needs_review": False,
            "issue": "",
        }

class RevitImporter:
    """
    Importer حرفه‌ای برای Scheduleهای جداگانه Revit / Dynamo.

    ویژگی‌ها:
    - TXT / CSV / TSV
    - CSV quoted
    - UTF-16 / UTF-8 / CP1256
    - تشخیص Dynamo semantic export
    - حفظ تمام ستون‌های اصلی در raw_data
    - عدم فرض Quantity=1
    - محاسبه صحیح وزن آرماتور
    - تشخیص category / structural group
    """

    def __init__(self):
        self.detector = RevitElementDetector()
        self.normalizer = TextNormalizer()

    # --------------------------------------------------------
    # Encoding
    # --------------------------------------------------------

    @staticmethod
    def detect_encoding(data: bytes) -> str:

        if not data:
            return "utf-8"

        if data.startswith(b"\xff\xfe") or data.startswith(b"\xfe\xff"):
            return "utf-16"

        candidates = [
            "utf-8-sig",
            "utf-8",
            "utf-16",
            "cp1256",
            "windows-1256",
            "latin1",
        ]

        for encoding in candidates:
            try:
                data.decode(encoding)
                return encoding
            except Exception:
                continue

        return "utf-8"

    # --------------------------------------------------------
    # Number helper
    # --------------------------------------------------------

    @staticmethod
    def safe_float(value: Any) -> Optional[float]:

        if value is None:
            return None

        if isinstance(value, (int, float)):
            try:
                value = float(value)

                if math.isnan(value) or math.isinf(value):
                    return None

                return value

            except Exception:
                return None

        text = str(value).strip()

        if not text:
            return None

        text = (
            text
            .replace(",", "")
            .replace("٬", "")
            .replace("،", ".")
        )

        try:
            value = float(text)

            if math.isnan(value) or math.isinf(value):
                return None

            return value

        except Exception:
            return None

    # --------------------------------------------------------
    # Header normalization
    # --------------------------------------------------------

    @staticmethod
    def normalize_header(value: Any) -> str:

        if value is None:
            return ""

        text = str(value).strip()

        text = text.replace("\ufeff", "")
        text = text.replace("\u200c", "")
        text = text.replace(" ", "")
        text = text.replace("-", "_")

        return text.lower()

    # --------------------------------------------------------
    # delimiter
    # --------------------------------------------------------

    def detect_delimiter(self, text: str) -> str:

        sample = "\n".join(text.splitlines()[:20])

        try:
            dialect = csv.Sniffer().sniff(
                sample,
                delimiters="\t,;"
            )

            return dialect.delimiter

        except Exception:
            if "\t" in sample:
                return "\t"

            if ";" in sample:
                return ";"

            return ","

    # --------------------------------------------------------
    # table extraction
    # --------------------------------------------------------

    def _read_table(
        self,
        text: str,
    ) -> List[List[str]]:

        delimiter = self.detect_delimiter(text)

        reader = csv.reader(
            io.StringIO(text),
            delimiter=delimiter,
            quotechar='"',
        )

        rows = []

        for row in reader:

            if not row:
                continue

            clean = [
                str(cell).strip()
                for cell in row
            ]

            if not any(clean):
                continue

            rows.append(clean)

        return rows

    # --------------------------------------------------------
    # header detection
    # --------------------------------------------------------

    def _find_header_row(
        self,
        rows: List[List[str]],
    ) -> int:

        if not rows:
            return 0

        best_idx = 0
        best_score = -1

        important_tokens = {
            "category",
            "family",
            "type",
            "mark",
            "volume",
            "area",
            "length",
            "quantity",
            "diameter_mm",
            "elementid",
            "uniqueid",
            "hostcategory",
        }

        for idx, row in enumerate(rows[:30]):

            score = 0

            normalized = {
                self.normalize_header(x)
                for x in row
            }

            for token in important_tokens:
                if token in normalized:
                    score += 1

            if score > best_score:
                best_score = score
                best_idx = idx

        return best_idx

    # --------------------------------------------------------
    # Dynamo detection
    # --------------------------------------------------------

    def _is_dynamo(
        self,
        headers: List[str],
    ) -> bool:

        h = {
            self.normalize_header(x)
            for x in headers
        }

        return (
            "category" in h
            and (
                "diameter_mm" in h
                or "hostcategory" in h
                or "structuralcontext" in h
                or "rebarshape" in h
                or "hostuniqueid" in h
            )
        )

    # --------------------------------------------------------
    # get field
    # --------------------------------------------------------

    @staticmethod
    def _get(
        row: Dict[str, Any],
        *keys: str,
    ) -> Any:

        normalized = {
            str(k).replace("\ufeff", "").strip().lower(): v
            for k, v in row.items()
        }

        for key in keys:
            key_norm = (
                str(key)
                .replace("\ufeff", "")
                .strip()
                .lower()
            )

            if key_norm in normalized:
                return normalized[key_norm]

        return None

    # --------------------------------------------------------
    # structural group
    # --------------------------------------------------------

    def _detect_structural_group(
        self,
        text: str,
        category: str = "",
    ) -> str:

        text_all = (
            f"{text} {category}"
        ).strip()

        try:
            return self.detector.detect_structural_group(
                text_all
            )
        except Exception:
            pass

        normalized = self.normalizer.normalize(
            text_all
        )

        checks = [
            ("foundation", ["foundation", "پی", "فونداسیون"]),
            ("column", ["column", "ستون"]),
            ("beam", ["beam", "تیر"]),
            ("slab", ["slab", "دال", "سقف"]),
            ("wall", ["wall", "دیوار"]),
            ("stair", ["stair", "پله"]),
            ("joist", ["joist", "تیرچه"]),
        ]

        for group, tokens in checks:

            for token in tokens:

                if token in normalized:
                    return group

        return ""

    # --------------------------------------------------------
    # category
    # --------------------------------------------------------

    def _detect_category(
        self,
        row: Dict[str, Any],
        raw_text: str,
    ) -> str:

        category = self._get(
            row,
            "Category",
            "category",
        )

        if category:
            return str(category).strip()

        return self.normalizer.detect_category(
            raw_text
        )

    # --------------------------------------------------------
    # technical tokens
    # --------------------------------------------------------

    def _tokens(
        self,
        text: str,
    ) -> Set[str]:

        text = self._norm(text)

        if not text:
            return set()

        return {
            token
            for token in re.split(
                r"[^a-z0-9آ-ی]+",
                text
            )
            if len(token) >= 2
        }

    # --------------------------------------------------------
    # description similarity
    # --------------------------------------------------------

    def _similarity(
        self,
        a: str,
        b: str,
    ) -> float:

        a_n = self._norm(a)
        b_n = self._norm(b)

        if not a_n or not b_n:
            return 0.0

        seq = difflib.SequenceMatcher(
            None,
            a_n,
            b_n,
        ).ratio()

        at = self._tokens(a_n)
        bt = self._tokens(b_n)

        overlap = (
            len(at & bt) /
            max(
                1,
                len(at | bt)
            )
        )

        return (
            seq * 0.55
            + overlap * 0.45
        ) * 100.0

    # candidate score
    # --------------------------------------------------------

    def _score_candidate(
        self,

        candidate: Dict[str, Any],
    ) -> Tuple[float, List[str]]:

        score = 0.0
        reasons = []

        source_text = " ".join(
            [
                self._clean(
                    item.get(
                        "raw_name",
                        ""
                    )
                ),
                self._clean(
                    item.get(
                        "element_type",
                        ""
                    )
                ),
                self._clean(
                    item.get(
                        "category",
                        ""
                    )
                ),
                self._clean(
                    item.get(
                        "structural_group",
                        ""
                    )
                ),
                self._clean(
                    item.get(
                        "material",
                        ""
                    )
                ),
            ]
        )

        target_text = self._clean(
            candidate.get(
                "description",
                ""
            )
        )

        # شرح
        description_score = self._similarity(
            source_text,
            target_text,
        )

        score += min(
            10.0,
            description_score * 0.10
        )

        if description_score >= 70:
            reasons.append(
                "شباهت شرح بالا"
            )

        elif description_score >= 45:
            reasons.append(
                "شباهت شرح متوسط"
            )

        # واحد
        source_unit = self._expected_unit(
            item
        )

        target_unit = candidate.get(
            "unit",
            ""
        )

        if source_unit and UnitEngine.compatible(
            source_unit,
            target_unit,
        ):
            score += 10.0
            reasons.append(
                f"واحد سازگار: {source_unit}"
            )

        elif source_unit:
            score -= 25.0
            reasons.append(
                "واحد ناسازگار"
            )

        # فصل
        candidate_chapter = self._clean(
            candidate.get(
                "chapter",
                ""
            )
        )

        candidate_chapter = (
            candidate_chapter[:2]
        )

        expected_chapters = (
            self._expected_chapters(
                item
            )
        )

        if (
            candidate_chapter
            and candidate_chapter
            in expected_chapters
        ):
            score += 15.0
            reasons.append(
                f"فصل سازگار: {candidate_chapter}"
            )

        # قطر
        diameter = item.get(
            "diameter",
            None
        )

        if diameter:

            description = self._clean(
                candidate.get(
                    "description",
                    ""
                )
            )

            diameter_texts = [
                str(int(diameter)),
                f"{int(diameter)}mm",
                f"قطر{int(diameter)}",
                f"φ{int(diameter)}",
            ]

            if any(
                token.lower()
                in description.lower()
                for token in diameter_texts
            ):
                score += 15.0
                reasons.append(
                    "قطر با شرح فهرست بها سازگار است"
                )

        # structural group
        group = self._norm(
            item.get(
                "structural_group",
                ""
            )
        )

        if group:

            candidate_text = self._norm(
                target_text
            )

            group_map = {
                "foundation": [
                    "پی",
                    "فونداسیون",
                    "foundation",
                ],
                "column": [
                    "ستون",
                    "column",
                ],
                "beam": [
                    "تیر",
                    "beam",
                ],
                "slab": [
                    "دال",
                    "سقف",
                    "slab",
                ],
                "wall": [
                    "دیوار",
                    "wall",
                ],
                "stair": [
                    "پله",
                    "stair",
                ],
            }

            for token in group_map.get(
                group,
                []
            ):
                if token.lower() in candidate_text.lower():
                    score += 8.0
                    reasons.append(
                        "گروه سازه‌ای سازگار"
                    )
                    break

        # material / technical
        technical_tokens = [
            "a1",
            "a2",
            "a3",
            "a4",
            "ribbed",
            "plain",
            "ساده",
            "آجدار",
            "آرماتور",
            "بتن",
            "قالب",
            "تیرچه",
            "بلوک",
        ]

        source_lower = self._norm(
            source_text
        )

        target_lower = self._norm(
            target_text
        )

        for token in technical_tokens:

            if (
                token in source_lower
                and token in target_lower
            ):
                score += 8.0
                reasons.append(
                    f"ویژگی فنی مشترک: {token}"
                )

        return max(
            0.0,
            min(100.0, score)
        ), reasons

    # --------------------------------------------------------
    # manual rule score
    # --------------------------------------------------------

    def _manual_match(
        self,
        item: Dict[str, Any],
        prices: pd.DataFrame,
        rules: pd.DataFrame,
    ) -> Optional[Dict[str, Any]]:

        if rules.empty:
            return None

        if "item_code" not in rules.columns:
            return None

        code = None

        item_text = self._norm(
            " ".join(
                [
                    self._clean(
                        item.get(
                            "raw_name",
                            ""
                        )
                    ),
                    self._clean(
                        item.get(
                            "category",
                            ""
                        )
                    ),
                    self._clean(
                        item.get(
                            "element_type",
                            ""
                        )
                    ),
                    self._clean(
                        item.get(
                            "structural_group",
                            ""
                        )
                    ),
                ]
            )
        )

        for _, rule in rules.iterrows():

            active = rule.get(
                "active",
                True
            )

            if str(active).lower() in (
                "false",
                "0",
                "no",
            ):
                continue

            revit_key = self._norm(
                rule.get(
                    "revit_key",
                    ""
                )
            )

            if not revit_key:
                continue

            if (
                revit_key in item_text
                or item_text in revit_key
            ):
                code = self._clean(
                    rule.get(
                        "item_code",
                        ""
                    )
                )

                if code:
                    break

        if not code:
            return None

        matches = prices[
            prices["item_code"]
            .astype(str)
            .str.strip()
            == str(code).strip()
        ]

        if matches.empty:
            return None

        candidate = (
            matches.iloc[0]
            .to_dict()
        )

        source_unit = self._expected_unit(
            item
        )

        if source_unit and not UnitEngine.compatible(
            source_unit,
            candidate.get(
                "unit",
                ""
            ),
        ):
            return {
                "status": "REVIEW",
                "score": 50.0,
                "item_code": str(code),
                "candidate": candidate,
                "reasons": [
                    "Mapping دستی وجود دارد اما واحد ناسازگار است"
                ],
                "method": "manual_rule_invalid_unit",
            }

        return {
            "status": "CONFIRMED",
            "score": 96.0,
            "item_code": str(
                candidate.get(
                    "item_code",
                    ""
                )
            ),
            "candidate": candidate,
            "reasons": [
                "Mapping دستی معتبر",
                "کد در فهرست‌بهای واقعی وجود دارد",
                "واحد سازگار است",
            ],
            "method": "manual_rule",
        }

    # --------------------------------------------------------
    # match one
    # --------------------------------------------------------

    def match_item(
        self,
        item: Dict[str, Any],
    ) -> Dict[str, Any]:

        prices = self._load_price_list()
        rules = self._load_mapping_rules()

        if prices.empty:

            return {
                "item_code": "",
                "item_description": "",
                "unit": "",
                "unit_price": 0.0,
                "match_method": "no_pricebook",
                "match_confidence": 0.0,
                "match_status": "UNMATCHED",
                "match_details": "فهرست بهای ۱۴۰۴ در پایگاه داده موجود نیست",
                "needs_review": True,
                "issues": "بدون فهرست بها",
            }

        manual = self._manual_match(
            item,
            prices,
            rules,
        )

        if manual is not None:

            candidate = manual["candidate"]

            return {
                "item_code": str(
                    candidate.get(
                        "item_code",
                        ""
                    )
                ),
                "item_description": str(
                    candidate.get(
                        "description",
                        ""
                    )
                ),
                "unit": str(
                    candidate.get(
                        "unit",
                        ""
                    )
                ),
                "unit_price": float(
                    candidate.get(
                        "unit_price",
                        0
                    ) or 0
                ),
                "match_method": manual["method"],
                "match_confidence": manual["score"],
                "match_status": manual["status"],
                "match_details": " | ".join(
                    manual["reasons"]
                ),
                "needs_review": (
                    manual["status"] != "CONFIRMED"
                ),
                "issues": "",
            }

        candidates = []

        for _, row in prices.iterrows():

            candidate = row.to_dict()

            score, reasons = (
                self._score_candidate(
                    item,
                    candidate
                )
            )

            candidates.append(
                (
                    score,
                    candidate,
                    reasons,
                )
            )

        if not candidates:

            return {
                "item_code": "",
                "item_description": "",
                "unit": "",
                "unit_price": 0.0,
                "match_method": "none",
                "match_confidence": 0.0,
                "match_status": "UNMATCHED",
                "match_details": "کاندیدایی پیدا نشد",
                "needs_review": True,
                "issues": "عدم تطبیق",
            }

        candidates.sort(
            key=lambda x: x[0],
            reverse=True,
        )

        best_score, best, reasons = (
            candidates[0]
        )

        second_score = (
            candidates[1][0]
            if len(candidates) > 1
            else 0.0
        )

        margin = best_score - second_score

        if (
            best_score
            >= self.CONFIRMED_THRESHOLD
            and margin
            >= self.MIN_MARGIN
        ):
            status = "CONFIRMED"

        elif (
            best_score
            >= self.REVIEW_THRESHOLD
        ):
            status = "REVIEW"

        else:
            status = "UNMATCHED"

        return {
            "item_code": str(
                best.get(
                    "item_code",
                    ""
                )
            ),
            "item_description": str(
                best.get(
                    "description",
                    ""
                )
            ),
            "unit": str(
                best.get(
                    "unit",
                    ""
                )
            ),
            "unit_price": float(
                best.get(
                    "unit_price",
                    0
                )
                or 0
            ),
            "match_method": "semantic_fuzzy",
            "match_confidence": float(
                best_score
            ),
            "match_status": status,
            "match_details": (
                " | ".join(reasons)
                + f" | فاصله از گزینه دوم: {margin:.1f}"
            ),
            "needs_review": (
                status != "CONFIRMED"
            ),
            "issues": (
                ""
                if status == "CONFIRMED"
                else "نیازمند بررسی کارشناسی"
            ),
        }

    # --------------------------------------------------------
    # dataframe matching
    # --------------------------------------------------------

    def match_dataframe(
        self,
        df: pd.DataFrame,
    ) -> pd.DataFrame:

        if df is None or df.empty:
            return df.copy()

        rows = []

        for _, row in df.iterrows():

            base = row.to_dict()

            match = self.match_item(
                base
            )

            base.update(match)

            rows.append(base)

        return pd.DataFrame(rows)

class TakeoffEngine:

    def __init__(
        self,
        db,
        project_id: Optional[int] = None,
    ):
        self.db = db
        self.project_id = project_id

        self.mapping_engine = MappingEngine(
            db=db,
            project_id=project_id,
        )

    # --------------------------------------------------------
    # helpers
    # --------------------------------------------------------

    @staticmethod
    def _safe_num(
        value: Any,
    ) -> float:

        try:

            if value is None:
                return 0.0

            text = str(value).strip()

            if not text:
                return 0.0

            value = float(
                text.replace(",", "")
            )

            if math.isnan(value) or math.isinf(value):
                return 0.0

            return value

        except Exception:
            return 0.0

    # --------------------------------------------------------
    # quantity
    # --------------------------------------------------------

    def _raw_quantity(
        self,
        row: Dict[str, Any],
        mapped_unit: str = "",
    ) -> Tuple[float, str, List[str]]:

        issues = []

        category = str(
            row.get(
                "category",
                ""
            )
        ).lower()

        is_rebar = (
            "rebar" in category
            or "reinforcement" in category
            or "آرماتور" in category
            or "میلگرد" in category
            or self._safe_num(
                row.get(
                    "diameter",
                    0
                )
            ) > 0
        )

        # ----------------------------
        # Rebar
        # ----------------------------

        if is_rebar:

            result = (
                RebarQuantityEngine.calculate(
                    row=row,
                    diameter=self._safe_num(
                        row.get(
                            "diameter"
                        )
                    ),
                )
            )

            if result["weight_kg"] > 0:

                return (
                    result["weight_kg"],
                    "kg",
                    (
                        [result["issue"]]
                        if result["issue"]
                        else []
                    ),
                )

            if result["needs_review"]:

                issues.append(
                    result["issue"]
                    or "مقدار آرماتور قابل محاسبه نیست"
                )

                return (
                    0.0,
                    "kg",
                    issues,
                )

        # ----------------------------
        # Concrete
        # ----------------------------

        if (
            "concrete" in category
            or "بتن" in category
        ):

            volume = self._safe_num(
                row.get("volume")
            )

            if volume > 0:
                return (
                    volume,
                    "m3",
                    [],
                )

            issues.append(
                "حجم بتن صفر یا نامشخص است"
            )

            return (
                0.0,
                "m3",
                issues,
            )

        # ----------------------------
        # Formwork
        # ----------------------------

        if (
            "formwork" in category
            or "قالب" in category
        ):

            area = self._safe_num(
                row.get("area")
            )

            if area > 0:
                return (
                    area,
                    "m2",
                    [],
                )

            issues.append(
                "سطح قالب‌بندی مشخص نیست"
            )

            return (
                0.0,
                "m2",
                issues,
            )

        # ----------------------------
        # Masonry
        # ----------------------------

        if (
            "masonry" in category
            or "بنایی" in category
            or "آجر" in category
            or "بلوک" in category
        ):

            area = self._safe_num(
                row.get("area")
            )

            if area > 0:
                return (
                    area,
                    "m2",
                    [],
                )

            volume = self._safe_num(
                row.get("volume")
            )

            if volume > 0:
                return (
                    volume,
                    "m3",
                    [],
                )

            issues.append(
                "مقدار بنایی مشخص نیست"
            )

            return (
                0.0,
                "",
                issues,
            )

        # ----------------------------
        # General fallback
        # ----------------------------

        for field_name, unit in (
            ("volume", "m3"),
            ("area", "m2"),
            ("length", "m"),
            ("count", "ea"),
            ("weight", "kg"),
        ):

            value = self._safe_num(
                row.get(
                    field_name
                )
            )

            if value > 0:

                return (
                    value,
                    unit,
                    [],
                )

        issues.append(
            "کمیت قابل استفاده برای متره پیدا نشد"
        )

        return (
            0.0,
            "",
            issues,
        )

    # --------------------------------------------------------
    # waste
    # --------------------------------------------------------

    def _waste_percent(
        self,
        category: str,
    ) -> float:

        try:

            method = getattr(
                self.db,
                "get_waste_settings",
                None
            )

            if callable(method):

                settings = method(
                    self.project_id
                )

                if isinstance(
                    settings,
                    dict
                ):

                    category_key = category.lower()

                    if (
                        category_key
                        in settings
                    ):
                        return float(
                            settings[
                                category_key
                            ]
                        )

        except Exception:
            pass

        category_lower = (
            category.lower()
        )

        if (
            "rebar" in category_lower
            or "reinforcement" in category_lower
            or "آرماتور" in category_lower
            or "میلگرد" in category_lower
        ):
            return DEFAULT_WASTE_PERCENTS[
                "rebar"
            ]

        if (
            "concrete"
            in category_lower
            or "بتن" in category_lower
        ):
            return DEFAULT_WASTE_PERCENTS[
                "concrete"
            ]

        if (
            "formwork"
            in category_lower
            or "قالب" in category_lower
        ):
            return DEFAULT_WASTE_PERCENTS[
                "formwork"
            ]

        return DEFAULT_WASTE_PERCENTS[
            "default"
        ]

    # --------------------------------------------------------
    # coefficient
    # --------------------------------------------------------

    def _coefficient(
        self,
        name: str,
    ) -> float:

        defaults = {
            "taadil": 1.0,
            "balasari": 1.30,
        }

        try:

            method = getattr(
                self.db,
                "get_coefficients",
                None
            )

            if callable(method):

                data = method(
                    self.project_id
                )

                if isinstance(
                    data,
                    dict
                ):

                    value = data.get(
                        name
                    )

                    if value is not None:
                        return float(
                            value
                        )

        except Exception:
            pass

        return defaults.get(
            name,
            1.0
        )

    # --------------------------------------------------------
    # main processing
    # --------------------------------------------------------

    def process(
        self,
        revit_data: List[Dict[str, Any]],
    ) -> Dict[str, Any]:

        if not revit_data:

            return {
                "items": [],
                "rebar_listof": [],
                "quality": {
                    "total_rows": 0,
                    "valid_rows": 0,
                    "confirmed": 0,
                    "review": 0,
                    "unmatched": 0,
                    "zero_quantity": 0,
                    "no_price": 0,
                    "exact": 0,
                    "semantic": 0,
                    "manual": 0,
                    "total_items": 0,
                    "match_rate": 0.0,
                    "exact_rate": 0.0,
                    "review_rate": 0.0,
                    "unmatched_rate": 0.0,
                    "no_price_rate": 0.0,
                },
            }

        matched_rows = []
        rebar_listof = []

        confirmed = 0
        review = 0
        unmatched = 0
        zero_quantity = 0
        no_price = 0
        exact = 0
        semantic = 0
        manual = 0
        valid_rows = 0

        for index, row in enumerate(
            revit_data,
            start=1
        ):

            if not isinstance(
                row,
                dict
            ):
                continue

            valid_rows += 1

            raw_quantity, detected_unit, quantity_issues = (
                self._raw_quantity(
                    row
                )
            )

            mapping = (
                self.mapping_engine.match_item(
                    row
                )
            )

            item_unit = (
                mapping.get(
                    "unit",
                    ""
                )
                or detected_unit
            )

            raw_quantity_value = (
                float(raw_quantity)
            )

            if raw_quantity_value <= 0:
                zero_quantity += 1

            if not mapping.get(
                "item_code"

    @staticmethod
    def _ensure_columns(
        df: pd.DataFrame,
        columns: List[str]
    ) -> pd.DataFrame:
        """Ensure required columns exist without destroying existing data."""

        if df is None or df.empty:
            return pd.DataFrame(columns=columns)

        result = df.copy()

        for col in columns:
            if col not in result.columns:
                result[col] = ""

        return result

    @staticmethod
    def _safe_numeric(value: Any, default: float = 0.0) -> float:
        """Convert values safely for Excel output."""

        try:
            if value is None:
                return default

            if isinstance(value, bool):
                return float(value)

            if isinstance(value, str):
                value = value.strip()

                if not value:
                    return default

                value = (
                    value
                    .replace(",", "")
                    .replace("٬", "")
                    .replace("،", "")
                )

            number = float(value)

            if math.isnan(number) or math.isinf(number):
                return default

            return number

        except Exception:
            return default

    @staticmethod
    def _prepare_traceability(
        items_df: pd.DataFrame
    ) -> pd.DataFrame:
        """
        Prepare detailed rows while preserving complete Revit traceability.

        Important:
        - No quantity is artificially changed to 1.
        - Missing Quantity remains missing/zero according to source result.
        - All semantic identifiers are preserved when available.
        """

        if items_df is None or items_df.empty:
            return pd.DataFrame(columns=ExcelExporter.TRACE_COLUMNS)

        df = items_df.copy()

        for col in ExcelExporter.TRACE_COLUMNS:
            if col not in df.columns:
                df[col] = ""

        return df

    @staticmethod
    def _write_title(
        worksheet,
        title: str,
        title_fmt,
        column_count: int
    ) -> None:
        """Write merged title safely."""

        if column_count <= 0:
            column_count = 1

        worksheet.merge_range(
            0,
            0,
            0,
            column_count - 1,
            title,
            title_fmt,
        )

    @staticmethod
    def _format_sheet(
        worksheet,
        dataframe: pd.DataFrame,
        header_fmt,
        start_row: int,
        default_width: int = 15
    ) -> None:
        """Apply professional formatting to an Excel worksheet."""

        for i, col in enumerate(dataframe.columns):
            worksheet.write(
                start_row,
                i,
                col,
                header_fmt,
            )

            column_name = str(col)

            if "شرح" in column_name:
                width = 45
            elif "نام" in column_name:
                width = 35
            elif "جزئیات" in column_name:
                width = 45
            elif "مشکلات" in column_name:
                width = 35
            elif "فایل" in column_name:
                width = 28
            elif "جدول" in column_name:
                width = 28
            elif "UniqueId" in column_name:
                width = 32
            elif "ElementId" in column_name:
                width = 16
            else:
                width = default_width

            worksheet.set_column(
                i,
                i,
                width,
            )

        worksheet.freeze_panes(
            start_row + 1,
            0,
        )

        worksheet.autofilter(
            start_row,
            0,
            max(start_row, start_row + len(dataframe)),
            max(0, len(dataframe.columns) - 1),
        )

class MappingImporter:
    EXPECTED_COLUMNS = ["revit_key", "match_type", "usage", "category", "chapter",
                        "item_code", "size", "rebar_dia", "notes", "priority", "active"]

    @classmethod
    def import_excel(cls, file_data: io.BytesIO, filename: str = "") -> Tuple[pd.DataFrame, Dict]:
        info = {"filename": filename, "rows": 0, "errors": [], "warnings": []}
        try: df = pd.read_excel(file_data, engine="openpyxl", dtype=str)
        except Exception as e:
            info["errors"].append(f"خطا: {e}")
            return pd.DataFrame(), info
        if df.empty:
            info["errors"].append("فایل خالی است")
            return df, info

        col_map = {}
        for col in df.columns:
            norm = TextNormalizer.normalize(col).replace(" ", "_")
            for expected in cls.EXPECTED_COLUMNS:
                if norm == expected or col.strip().lower() == expected:
                    col_map[col] = expected
                    break
            if col not in col_map:
                best = FuzzyMatcher.best_match(norm, cls.EXPECTED_COLUMNS, threshold=0.7)
                if best: col_map[col] = best[0]

        df = df.rename(columns=col_map)
        for col in cls.EXPECTED_COLUMNS:
            if col not in df.columns: df[col] = ""

        df["active"] = df["active"].apply(lambda x: 1 if str(x).strip().lower() in ("1", "true", "بله", "yes", "") else 0)
        df["priority"] = df["priority"].apply(lambda x: int(TextNormalizer.safe_float(x)))
        info["rows"] = len(df)
        return df, info

def page_settings() -> None:
    st.markdown("# ⚙️ تنظیمات پروژه")
    pid = _require_project()
    if not pid: return
    db = get_db()

    st.markdown("### 📊 درصد پرت مصالح")
    waste = db.get_waste_settings(pid)
    waste_labels = {"concrete": "بتن", "rebar": "آرماتور", "formwork": "قالب‌بندی",
                    "block_joist": "سقف تیرچه بلوک", "masonry": "بنایی", "default": "سایر (پیش‌فرض)"}
    changed = False
    cols = st.columns(3)
    for i, (k, label) in enumerate(waste_labels.items()):
        with cols[i % 3]:
            cur = waste.get(k, DEFAULT_WASTE_PERCENTS.get(k, 3.0))
            new = st.number_input(f"{label} (%)", min_value=0.0, max_value=50.0,
                                   value=float(cur), step=0.5, key=f"w_{k}")
            if new != cur:
                db.update_waste_setting(pid, k, new)
                changed = True
    if changed: st.success("✅ درصد پرت به‌روزرسانی شد")

    st.markdown("---")
    st.markdown("### 📐 ضرایب پیمان")
    coeffs = db.get_coefficients(pid)
    changed2 = False
    cols2 = st.columns(3)
    for i, (name, data) in enumerate(coeffs.items()):
        with cols2[i % 3]:
            new = st.number_input(data["description"], min_value=0.0, max_value=10.0,
                                   value=float(data["value"]), step=0.01, key=f"c_{name}")
            if new != data["value"]:
                db.update_coefficient(pid, name, new)
                changed2 = True
    if changed2: st.success("✅ ضرایب به‌روزرسانی شد")

    result = st.session_state.get(

    )

            "last_result"
        )

    if not result:
        st.info(
            "ابتدا پردازش متره را اجرا کنید."
        )
        return

    # ---------------------------------------------------------
    # داده‌های نتیجه
    # ---------------------------------------------------------

        "items_df"
    )

    if items_df is None:

        items = result.get(
            "items",
            []
        )

        if items:
            items_df = pd.DataFrame(
                items
            )
        else:
            items_df = make_empty_df(
                MATCHED_ITEM_SCHEMA
            )

    if not isinstance(
        items_df,
        pd.DataFrame
    ):
        items_df = pd.DataFrame(
            items_df
        )

    summary = result.get(
        "summary",
        {}
    ) or {}

    quality = result.get(
        "quality",
        {}
    ) or {}

    # ---------------------------------------------------------
    # ضرایب
    # ---------------------------------------------------------

    db = get_db()

    try:
        coeffs = db.get_coefficients(
            pid
        )
    except Exception:
        coeffs = {}

    total = summary.get(
        "total_price",
        0
    ) or 0

    final = float(total)

    if coeffs:

        for name, data in coeffs.items():

            try:
                value = float(
                    data.get(
                        "value",
                        1
                    )
                )
            except Exception:
                value = 1.0

            final *= value

    # ---------------------------------------------------------
    # خلاصه مالی
    # ---------------------------------------------------------

    st.markdown(
        "### 💰 خلاصه مالی"
    )

    c1, c2, c3, c4 = st.columns(4)

    with c1:
        st.metric(
            "مبلغ خام (ریال)",
            f"{float(total):,.0f}"
        )

    with c2:
        st.metric(
            "مبلغ با ضرایب (ریال)",
            f"{float(final):,.0f}"
        )

    with c3:
        st.metric(
            "تعداد ردیف نهایی",

        )

    with c4:
        st.metric(
            "نرخ تطبیق",
            f"{float(quality.get('match_rate', 0) or 0):.1f}%"
        )

    # ---------------------------------------------------------
    # تفکیک بر اساس دسته
    # ---------------------------------------------------------

    st.markdown(
        "### 🏗️ تفکیک بر اساس دسته"
    )

    cat_data = summary.get(
        "by_category",
        {}
    )

    if cat_data:

        cat_rows = []

        for category, data in cat_data.items():

            if not isinstance(
                data,
                dict
            ):
                continue

            cat_rows.append({
                "دسته": category,
                "تعداد": data.get(
                    "count",
                    0
                ),
                "مبلغ (ریال)": float(
                    data.get(
                        "total_price",
                        0
                    ) or 0
                ),
            })

        if cat_rows:

            cdf = pd.DataFrame(
                cat_rows
            )

            st.dataframe(
                cdf,
                use_container_width=True,
                hide_index=True,
            )

    else:
        st.info(
            "اطلاعات تفکیک دسته‌ای موجود نیست."
        )

    # ---------------------------------------------------------
    # تفکیک بر اساس فصل
    # ---------------------------------------------------------

    st.markdown(
        "### 📚 تفکیک بر اساس فصل"
    )

    ch_data = summary.get(
        "by_chapter",
        {}
    )

    if ch_data:

        chapter_rows = []

        for chapter, data in ch_data.items():

            if not isinstance(
                data,
                dict
            ):
                continue

            chapter_rows.append({
                "فصل": chapter,
                "تعداد": data.get(
                    "count",
                    0
                ),
                "مبلغ (ریال)": float(
                    data.get(
                        "total_price",
                        0
                    ) or 0
                ),
            })

        if chapter_rows:

            chdf = pd.DataFrame(
                chapter_rows
            )

            st.dataframe(
                chdf,
                use_container_width=True,
                hide_index=True,
            )

    else:
        st.info(
            "اطلاعات تفکیک فصلی موجود نیست."
        )

    # ---------------------------------------------------------
    # صورت وضعیت تجمیعی
    # ---------------------------------------------------------

    st.markdown(
        "### 📑 صورت وضعیت تجمیعی"
    )

    if items_df.empty:

        st.info(
            "آیتم نهایی برای نمایش وجود ندارد."
        )
        return

    work_df = items_df.copy()

    for col in [
        "item_code",
        "unit",
        "item_description",
        "final_quantity",
        "unit_price",
        "total_price",
    ]:

        if col not in work_df.columns:
            work_df[col] = ""

    valid = work_df[
        work_df["item_code"]
        .fillna("")
        .astype(str)
        .str.strip()
        != ""
    ].copy()

    if valid.empty:

        st.warning(
            "هیچ ردیفی با کد معتبر فهرست بها برای صورت وضعیت نهایی وجود ندارد."
        )
        return

    # ---------------------------------------------------------
    # تجمیع بر اساس کد + واحد
    # ---------------------------------------------------------

    agg = (
        valid.groupby(
            [
                "item_code",
                "unit",
            ],
            as_index=False,
            dropna=False,
        )
        .agg({
            "item_description": "first",
            "final_quantity": "sum",
            "unit_price": "first",
            "total_price": "sum",
        })
    )

    agg.columns = [
        "کد ردیف",
        "واحد",
        "شرح",
        "مقدار",
        "بهای واحد",
        "مبلغ کل",
    ]

        [
            "کد ردیف",
            "شرح",
            "واحد",
            "مقدار",
            "بهای واحد",
            "مبلغ کل",
        ]
    ]

    st.dataframe(
        agg,
        use_container_width=True,
        hide_index=True,
    )

    # دریافت نتیجه بدون اجرای مجدد
    # ---------------------------------------------------------

    result = st.session_state.get(
        "takeoff_result"
    )

    if not result:

            []
        )

    if not isinstance(
        rebar_df,

    ):
        rebar_df = pd.DataFrame(
            rebar_df
        )

    if rebar_df.empty:

        st.info(
            "ردیف آرماتوری یافت نشد."
        )
        return

    # ---------------------------------------------------------
    # اطمینان از وجود ستون‌های مورد نیاز

    required_columns = [
        "source_file",
        "source_table",

        "HostElementId",
        "location",
        "structural_group",
        "sub_group",
        "position",
        "diameter",
        "bar_length",
        "count",
        "net_weight",
        "waste_percent",
        "gross_weight",
        "item_code",
        "unit_price",
        "total_price",
        "match_status",
        "needs_review",
        "issues",
    ]

    for col in required_columns:

        if col not in rebar_df.columns:
            rebar_df[col] = ""

    # ---------------------------------------------------------
    # خلاصه وزن بر اساس گروه سازه‌ای
    # ---------------------------------------------------------

    st.markdown(
        "### 🏗️ خلاصه وزن آرماتور بر اساس گروه سازه‌ای"
    )

    group_df = rebar_df.copy()

    group_df[
        "net_weight"
    ] = pd.to_numeric(
        group_df[
            "net_weight"
        ],
        errors="coerce"
    ).fillna(0)

    group_df[
        "gross_weight"
    ] = pd.to_numeric(
        group_df[
            "gross_weight"
        ],
        errors="coerce"
    ).fillna(0)

    gs = (
        group_df.groupby(
            "structural_group",
            dropna=False
        )
        .agg({
            "net_weight": "sum",
            "gross_weight": "sum",
        })
        .reset_index()
    )

        "گروه سازه‌ای",
        "وزن خالص (kg)",
        "وزن با پرت (kg)",

        use_container_width=True,
        hide_index=True,
    )

        "net_weight"

        diameter_df[
            "net_weight"
        ],

        "gross_weight"

        diameter_df[

        "total_price",
        "match_status",
        "needs_review",
        "issues",
    ]

        c
        for c in display_columns

    disp = rebar_df[

    display_names = {

def page_quality() -> None:

    st.markdown("# ✅ گزارش کیفیت داده")

    # ---------------------------------------------------------
    # پروژه فعال
    # ---------------------------------------------------------
    pid = _require_project()

    if not pid:
        return

    # ---------------------------------------------------------
    # نتیجه آخرین پردازش
    #
    # نسخه جدید پردازش در takeoff_result ذخیره می‌شود.
    # برای سازگاری با نسخه قبلی، last_result نیز پشتیبانی می‌شود.

    if not result:

    )

    # برای سازگاری با ساختارهای قدیمی
    kpi = result.get(
        "kpi",
        {}
    )

    # KPI های اصلی
    # ---------------------------------------------------------
    st.markdown("### 📊 شاخص‌های کلیدی")

    c1, c2, c3, c4 = st.columns(4)

    match_rate = quality.get(
        "match_rate",
        kpi.get("match_rate", 0)
    )

    review_rate = quality.get(
        "review_rate",
        0
    )

    unmatched_rate = quality.get(
        "unmatched_rate",
        kpi.get("unmatched_rate", 0)
    )

    zero_quantity_rate = quality.get(
        "zero_quantity_rate",

    )

    total_rows = quality.get(

    valid_rows = quality.get(
        "valid_rows",
